### 1.1 SkillSystem.OnUpdate 双重调用

**问题**: `SkillSystem.Update()` 和 `GameManager.Update()` 同时调用 `OnUpdate(Time.deltaTime)`，导致冷却以2倍速递减。Q技能60s冷却实际30s耗尽。

**文件**: `Client/Assets/Scripts/Skill/SkillSystem.cs`

**当前代码**（问题代码）:
```csharp
private void Update()
{
    OnUpdate(Time.deltaTime);
}
```

**修复方案**: 删除 `SkillSystem.Update()` 方法，统一由 `GameManager.Update()` 驱动。

**操作步骤**:
1. 打开 `SkillSystem.cs`
2. 找到 `private void Update()` 方法
3. 整个方法删除
4. 确认 `GameManager.cs:Update()` 中已有 `SkillSystem.Instance.OnUpdate(dt)` 调用（当前在第83行，已存在）

**验证**: 运行游戏，释放Q技能后观察冷却时间是否为正确值（60-90s），而非减半。

```csharp
private bool _hasLanded = false;

private void Update()
{
    if (State == FragmentState.Falling && !_hasLanded)
    {
        // 检测是否落地
        if (_rb != null && _rb.velocity.y <= 0.01f)
        {
            // 使用射线检测是否接触地面
            RaycastHit2D hit = Physics2D.Raycast(
                transform.position, Vector2.down, 0.2f,
                1 << LayerMask.NameToLayer("Ground"));
            if (hit.collider != null)
            {
                _hasLanded = true;
                // 落地后停止物理模拟
                _rb.velocity = Vector2.zero;
                _rb.isKinematic = true;
            }
        }
        return; // 未落地前不启动倒计时
    }

    if (State == FragmentState.Falling && _hasLanded)
    {
        RemainingTime -= Time.deltaTime;
        if (RemainingTime <= 0f)
        {
            SetState(FragmentState.Despawned);
            if (FragmentSystem.HasInstance)
                FragmentSystem.Instance.OnFragmentDespawned(FragmentId);
        }
    }
}
```

### 2.3 FragmentSystem.GenerateDropPlan 分阶段生成

**问题**: `GenerateDropPlan` 一次性生成所有碎片，不分预告/收集阶段；忽略 `disasterType` 参数；硬编码基数28；DropTime 不符合阶段设计。

**文件**: `Client/Assets/Scripts/Fragment/FragmentSystem.cs`

**当前代码** (FragmentSystem.cs:193-215):
```csharp
public List<FragmentDropPlan> GenerateDropPlan(int disasterType, float densityFactor, uint seed)
{
    List<FragmentDropPlan> plan = new List<FragmentDropPlan>();
    System.Random rng = new System.Random((int)seed);

    int totalBase = Mathf.RoundToInt(28 * densityFactor);
    int totalCount = Mathf.Max(totalBase, 10);

    for (int i = 0; i < totalCount; i++)
    {
        FragmentType type = GenerateFragmentType(rng);
        Vector2 pos = GenerateDropPosition(rng);
        plan.Add(new FragmentDropPlan
        {
            FragmentId = _nextFragmentId++,
            Type = type,
            Position = pos,
            DropTime = i < 5 ? i * 1f : 5f + (i - 5) * 0.5f,
            Seed = (uint)rng.Next(),
        });
    }
    return plan;
}
```

**修复方案**: 拆分为预告阶段和收集阶段，使用 FragmentConfig 的 PreviewCount/CollectPhaseCount。

**替换为**:
```csharp
public List<FragmentDropPlan> GenerateDropPlan(int disasterType, float densityFactor, uint seed)
{
    List<FragmentDropPlan> plan = new List<FragmentDropPlan>();
    System.Random rng = new System.Random((int)seed);

    // 从配置读取预告和收集阶段的碎片数量
    int previewCount = _config != null ? _config.PreviewCount : 5;
    int collectCount = _config != null ? _config.CollectPhaseCount : 25;

    // 应用密度系数（3种碎片密度系数：1.0/0.85/0.7）
    collectCount = Mathf.RoundToInt(collectCount * densityFactor);
    collectCount = Mathf.Max(collectCount, 10);

    // 根据 disasterType 调整碎片类型分布权重
    // 火山类灾难偏向熔岩碎片，冰雪类偏向冰晶碎片
    int disasterCategory = disasterType / 100; // E=0, V=1, T=2, S=3, P=4, M=5

    // 预告阶段：3-5个碎片，1s间隔
    for (int i = 0; i < previewCount; i++)
    {
        plan.Add(new FragmentDropPlan
        {
            FragmentId = _nextFragmentId++,
            Type = GenerateFragmentType(rng, disasterCategory),
            Position = GenerateDropPosition(rng),
            DropTime = i * 1f, // 预告阶段每秒1个
            Seed = (uint)rng.Next(),
        });
    }

    // 收集阶段：20-30个碎片，0.5s间隔
    for (int i = 0; i < collectCount; i++)
    {
        plan.Add(new FragmentDropPlan
        {
            FragmentId = _nextFragmentId++,
            Type = GenerateFragmentType(rng, disasterCategory),
            Position = GenerateDropPosition(rng),
            DropTime = previewCount + i * 0.5f, // 收集阶段每0.5s一个
            Seed = (uint)rng.Next(),
        });
    }

    return plan;
}

/// <summary>
/// 根据灾难类别生成碎片类型。不同灾难类别影响碎片分布权重。
/// </summary>
private FragmentType GenerateFragmentType(System.Random rng, int disasterCategory)
{
    float roll = (float)rng.NextDouble();
    float iceProb, lavaProb;

    if (_config != null)
    {
        iceProb = _config.GetTypeProbability(FragmentType.IceCrystal);
        lavaProb = _config.GetTypeProbability(FragmentType.Lava);
    }
    else
    {
        iceProb = 0.55f;
        lavaProb = 0.30f;
    }

    // 根据灾难类别微调权重
    switch (disasterCategory)
    {
        case 0: // Element — 偏向冰晶和熔岩
            iceProb += 0.05f;
            lavaProb += 0.05f;
            break;
        case 1: // Environment — 均衡
            break;
        case 3: // Space — 偏向岩石
            iceProb -= 0.05f;
            lavaProb -= 0.05f;
            break;
    }

    if (roll < iceProb) return FragmentType.IceCrystal;
    if (roll < iceProb + lavaProb) return FragmentType.Lava;
    return FragmentType.Rock;
}
```

**同步修改**: 将原 `GenerateFragmentType(System.Random rng)` 方法重命名为 `GenerateFragmentType(System.Random rng, int disasterCategory)`，并更新所有调用处。

**验证**: 调用 `GenerateDropPlan` 后检查返回的 List，前几个碎片 DropTime 间隔1s（预告阶段），后续间隔0.5s（收集阶段），总数为 previewCount + collectCount。

### 2.4 FragmentSystem.GenerateDropPosition 读取配置

**问题**: 掉落位置硬编码 `x∈[-10,10], y∈[8,13]`，不读取 FragmentConfig.DropRangeMin/DropRangeMax。

**文件**: `Client/Assets/Scripts/Fragment/FragmentSystem.cs`

**当前代码** (FragmentSystem.cs:478-483):
```csharp
private Vector2 GenerateDropPosition(System.Random rng)
{
    float x = (float)(rng.NextDouble() * 20f - 10f);
    float y = (float)(rng.NextDouble() * 5f + 8f);
    return new Vector2(x, y);
}
```

**修复方案**: 从 FragmentConfig 读取范围，并参考建筑区域边界。

**替换为**:
```csharp
private Vector2 GenerateDropPosition(System.Random rng)
{
    float minX, maxX, minY, maxY;

    if (_config != null)
    {
        minX = _config.DropRangeMin;
        maxX = _config.DropRangeMax;
        minY = 8f;  // 天空高度
        maxY = 13f; // 天空高度上限
    }
    else
    {
        minX = -10f;
        maxX = 10f;
        minY = 8f;
        maxY = 13f;
    }

    // 尝试参考建筑区域边界，确保碎片不掉在建筑正上方
    IBuildSystem buildSys = ServiceLocator.Get<IBuildSystem>();
    if (buildSys != null && buildSys.Buildings.Count > 0)
    {
        // 获取建筑区域中心X，碎片偏向建筑区域两侧掉落
        float buildingCenterX = 0f;
        int count = 0;
        foreach (var b in buildSys.Buildings)
        {
            buildingCenterX += b.GridPosition.x;
            count++;
        }
        if (count > 0)
            buildingCenterX /= count;

        // 碎片掉落范围以建筑区域为中心，左右各扩展
        float halfRange = (maxX - minX) * 0.5f;
        minX = buildingCenterX - halfRange;
        maxX = buildingCenterX + halfRange;
    }

    float x = (float)(rng.NextDouble() * (maxX - minX) + minX);
    float y = (float)(rng.NextDouble() * (maxY - minY) + minY);
    return new Vector2(x, y);
}
```

**验证**: 调用 `GenerateDropPosition` 多次，检查返回值的 X 范围是否对应 FragmentConfig 的 DropRangeMin/DropRangeMax。

#### 2.5.1 碎片验证消耗

在 `TryStartSynthesis` 方法中，开始合成前检查角色携带的碎片类型和数量是否满足配方要求:

```csharp
public SynthesisRecipe? TryStartSynthesis(byte playerId, FragmentType fragmentType, MaterialType desiredOutput)
{
    // 检查合成台是否被占用
    // ...（现有合成台检查逻辑）

    // 获取配方
    SynthesisRecipe? recipe = FindRecipe(fragmentType, desiredOutput);
    if (!recipe.HasValue)
        return null;

    // 验证角色携带碎片
    ICharacterSystem charSys = ServiceLocator.Get<ICharacterSystem>();
    if (charSys == null) return null;

    CharacterController character = charSys.GetCharacter((CharacterType)playerId);
    if (character == null || character.Stats == null) return null;

    // 检查碎片数量是否足够
    int requiredCount = _m1ElementDepletion ? recipe.Value.RequiredCount * 2 : recipe.Value.RequiredCount;
    int carriedCount = CountCarriedFragmentsByType(character.Stats.CarriedFragmentIds, fragmentType);
    if (carriedCount < requiredCount)
    {
        Debug.Log($"[SynthesisSystem] 碎片不足: 需要{requiredCount}个{fragmentType}，实际携带{carriedCount}个");
        return null;
    }

    // 消耗碎片
    ConsumeFragmentsByType(character, fragmentType, requiredCount);

    // 开始合成计时
    // ...（现有计时逻辑）
    return recipe;
}

/// <summary>
/// 统计角色携带的指定类型碎片数量。
/// </summary>
private int CountCarriedFragmentsByType(List<int> fragmentIds, FragmentType type)
{
    int count = 0;
    IFragmentSystem fragSys = ServiceLocator.Get<IFragmentSystem>();
    if (fragSys == null) return 0;

    foreach (int id in fragmentIds)
    {
        if (fragSys.TryGetFragmentType(id, out FragmentType t) && t == type)
            count++;
    }
    return count;
}

/// <summary>
/// 从角色携带的碎片中消耗指定类型和数量的碎片。
/// </summary>
private void ConsumeFragmentsByType(CharacterController character, FragmentType type, int count)
{
    IFragmentSystem fragSys = ServiceLocator.Get<IFragmentSystem>();
    if (fragSys == null) return;

    int consumed = 0;
    var idsToRemove = new List<int>();

    foreach (int id in character.Stats.CarriedFragmentIds)
    {
        if (consumed >= count) break;
        if (fragSys.TryGetFragmentType(id, out FragmentType t) && t == type)
        {
            idsToRemove.Add(id);
            consumed++;
        }
    }

    foreach (int id in idsToRemove)
    {
        character.RemoveFragment(id);
    }
}
```

#### 2.5.2 打断时返还碎片

在 `InterruptSynthesis` 中实现碎片返还:

```csharp
public void InterruptSynthesis(byte playerId)
{
    if (!_synthesisTimers.ContainsKey(playerId))
        return;

    // 获取当前合成配方，返还碎片
    if (_activeRecipes.TryGetValue(playerId, out var recipeEntry))
    {
        SynthesisRecipe recipe = recipeEntry.Value;
        ICharacterSystem charSys = ServiceLocator.Get<ICharacterSystem>();
        if (charSys != null)
        {
            CharacterController character = charSys.GetCharacter((CharacterType)playerId);
            if (character != null && character.Stats != null)
            {
                int requiredCount = _m1ElementDepletion ? recipe.RequiredCount * 2 : recipe.RequiredCount;
                // 返还碎片ID（生成新的碎片ID）
                for (int i = 0; i < requiredCount; i++)
                {
                    int newId = GenerateReturnFragmentId();
                    character.AddFragment(newId);
                }
                Debug.Log($"[SynthesisSystem] 合成打断，返还{requiredCount}个{recipe.InputType}碎片");
            }
        }
        _activeRecipes.Remove(playerId);
    }

    _synthesisTimers.Remove(playerId);
}

private int _returnIdCounter = 100000;
private int GenerateReturnFragmentId()
{
    return _returnIdCounter++;
}
```

#### 2.6.1 材料消耗

在 `PlaceBuilding` 中添加材料消耗检查:

```csharp
public bool PlaceBuilding(BuildingType type, MaterialType material, Vector2Int gridPos, byte playerId)
{
    // 检查网格占用
    if (_grid.IsOccupied(gridPos.x, gridPos.y))
        return false;

    // 检查材料消耗
    ICharacterSystem charSys = ServiceLocator.Get<ICharacterSystem>();
    if (charSys == null) return false;

}

private IEnumerator PlaceBuildingCoroutine(BuildingType type, MaterialType material, Vector2Int gridPos, byte playerId)
{
    float placeTime = _config != null ? _config.PlaceTime : 0.5f;
    yield return new WaitForSeconds(placeTime);

    // 创建建筑
    int buildingId = _nextBuildingId++;
    int baseHP = GetBuildingHP(type);

    // 冰砖暴风雪环境 HP+50%
```

```csharp
public void DamageBuilding(int buildingId, float dps, ShelterEnvironment environment, float deltaTime)
{
    var building = _buildings.Find(b => b.BuildingId == buildingId);
    if (building == null) return;

    // 获取抗性系数（5种环境×6种材料）
    float resistanceCoeff = GetResistanceCoefficient(building.Material, environment);

    // 朝向检查：朝向错误时抗性降一级（伤害×1.5）
    float facingMultiplier = IsFacingCorrect(building) ? 1f : 1.5f;

    // 安全区减免50%
    float zoneMultiplier = building.IsInSafeZone ? 0.5f : 1f;

    // 计算最终伤害
    float damage = dps * resistanceCoeff * facingMultiplier * zoneMultiplier * deltaTime;

    building.CurrentHP -= Mathf.RoundToInt(damage);

    if (building.CurrentHP <= 0)
    {
        _grid.Vacate(building.GridPosition.x, building.GridPosition.y);
        _buildings.Remove(building);
        EventBus.Instance.Publish(new BuildingDestroyedEvent { buildingId = buildingId });
        Debug.Log($"[BuildingSystem] 建筑{buildingId}被摧毁");
    }
}

/// <summary>
/// 获取抗性系数。5种环境×6种材料的完整矩阵。
/// ★★★ 免疫(0×) | ★★ 强抗性(0.3×) | ★ 抗性(0.6×) | — 正常(1.0×) | ✗ 弱点(1.5×)
/// </summary>
private float GetResistanceCoefficient(MaterialType material, ShelterEnvironment environment)
{
    // 抗性矩阵 [环境, 材料]
    // 行: Volcano(0), Flood(1), Blizzard(2), Earthquake(3), Meteorite(4)
    // 列: WaterBrick(0), IceBrick(1), FireBrick(2), LavaBrick(3), StoneBrick(4), WarmBrick(5)
    float[,] matrix = new float[,]
    {
        // Volcano
        { 1.5f, 1.5f, 0f,   0.3f, 0.6f, 0.6f },
        // Flood
        { 0f,   0.3f, 1.5f, 1.5f, 0.6f, 0.6f },
        // Blizzard
        { 0.3f, 0f,   1.5f, 1.5f, 0.6f, 0.6f },
        // Earthquake
        { 1.0f, 1.0f, 1.0f, 1.0f, 0.3f, 0.6f },
        // Meteorite
        { 1.0f, 1.5f, 1.0f, 1.0f, 0.6f, 0.3f },
    };

    int envIdx = (int)environment;
    int matIdx = (int)material;

    if (envIdx < 0 || envIdx >= 5 || matIdx < 0 || matIdx >= 6)
        return 1.0f;

    return matrix[envIdx, matIdx];
}
```

```csharp
public void OnUpdate(float deltaTime)
{
    if (GameManager.HasInstance && GameManager.Instance.State.IsGameOver)
        return;

    // 阶段特异性能量逻辑
    // 建造阶段: 不消耗能量（建造阶段④不消耗能量）
    // 修整阶段: 仅恢复不消耗

    bool energyActive = _currentPhase == GamePhase.FragmentCollect

    GetCharacterPositions(out aquaPos, out ignisPos);

        ? _params.FragmentCollectDistance
        : _params.ShelterDistance;

        float consumptionRate = _isFragmentCollectPhase
            ? _params.FragmentCollectConsumptionRate
            : _params.ConsumptionRate;
        UpdateEnergy(ref _aquaEnergy, inRange, deltaTime, consumptionRate);
        UpdateEnergy(ref _ignisEnergy, inRange, deltaTime, consumptionRate);

        UpdateBufferAndDamage(
            _aquaEnergy, ref _aquaBuffering, ref _aquaBufferTimer,

            _ignisEnergy, ref _ignisBuffering, ref _ignisBufferTimer,
            deltaTime, CharacterType.Ignis, false, ignisPos);
    }
    else
    {
        // 仅恢复不消耗（建造/修整/升级阶段）
        UpdateEnergy(ref _aquaEnergy, true, deltaTime, 0f);
        UpdateEnergy(ref _ignisEnergy, true, deltaTime, 0f);
    }
}
```

### 2.8 ShelterSystem 地震冲击波机制

**问题**: 地震按连续DPS处理（每帧扣血），应为每冲击波-3HP离散事件。

**文件**: `Client/Assets/Scripts/Shelter/ShelterSystem.cs`

**修复方案**: 为地震环境添加冲击波计时器。

```csharp
// 在类中添加字段
private float _earthquakeShockwaveTimer = 0f;
private const float EARTHQUAKE_SHOCKWAVE_INTERVAL = 2f; // 每2秒一次冲击波
private const int EARTHQUAKE_DAMAGE_PER_WAVE = 3; // 每次冲击波-3HP

// 在 UpdateBufferAndDamage 方法中，地震环境特殊处理
private void UpdateBufferAndDamage(
    float energy, ref bool buffering, ref float bufferTimer,
    float dt, CharacterType type, bool isAqua, Vector2 charPos)
{
    if (energy > 0f)
    {
        buffering = false;
        bufferTimer = 0f;
        return;
    }

    if (!buffering)
    {
        buffering = true;
        bufferTimer = _params.BufferTime;
        return;
    }

    bufferTimer -= dt;
    if (bufferTimer > 0f)
        return;

    // 地震环境：冲击波离散扣血
    if (CurrentEnvironment == ShelterEnvironment.Earthquake)
    {
        _earthquakeShockwaveTimer += dt;
        if (_earthquakeShockwaveTimer >= EARTHQUAKE_SHOCKWAVE_INTERVAL)
        {
            _earthquakeShockwaveTimer = 0f;

            // 冲击波伤害
            float multiplier = _params.DamageMultiplier;
            int currentHP = isAqua ? _aquaHP : _ignisHP;
            if (currentHP <= _dyingProtectThreshold)
                multiplier *= (1f - _dyingProtectReduction);

            // 建筑区域内50%减免
            if (_currentPhase == GamePhase.DisasterImpact && IsInBuildingZone(charPos))
                multiplier *= 0.5f;

            int damage = Mathf.CeilToInt(EARTHQUAKE_DAMAGE_PER_WAVE * multiplier);
            if (damage > 0)
                DealDamage(type, damage);
        }
        return;
    }

    // 陨石环境：概率判定
    if (CurrentEnvironment == ShelterEnvironment.Meteorite)
    {
        // ...（保持现有陨石逻辑，但添加距离概率）

        // 掩体内0%
        if (IsInBuildingZone(charPos))
            return;

        // 距离陨石落点越近概率越高
        IDisasterSystem disasterSys = ServiceLocator.Get<IDisasterSystem>();
        if (disasterSys != null)
        {
            Vector2 disasterPos = disasterSys.GetDisasterPosition();
            float distToDisaster = Vector2.Distance(charPos, disasterPos);

            float meteoriteChance;
            if (distToDisaster <= 1f)
                meteoriteChance = 0.9f;  // 1格内90%
            else if (distToDisaster <= 3f)
                meteoriteChance = 0.25f; // 3格内25%
            else
                meteoriteChance = 0.05f; // 远距离5%

            if (Random.Range(0f, 1f) >= meteoriteChance)
                return;
        }
        else
        {
            // 无灾难位置信息时使用50%基础概率
            if (Random.Range(0f, 1f) >= 0.5f)
                return;
        }

        // 扣血
        float damageRate = GetEnvironmentDamageRate(type);
        if (damageRate <= 0f) return;

        float multiplier = _params.DamageMultiplier;
        int currentHP = isAqua ? _aquaHP : _ignisHP;
        if (currentHP <= _dyingProtectThreshold)
            multiplier *= (1f - _dyingProtectReduction);

        if (_currentPhase == GamePhase.DisasterImpact && IsInBuildingZone(charPos))
            multiplier *= 0.5f;

        float damage = damageRate * multiplier * dt;
        int intDamage = Mathf.CeilToInt(damage);
        if (intDamage > 0)
            DealDamage(type, intDamage);

        return;
    }

    // 其他环境：连续DPS（保持现有逻辑）
    float baseDamageRate = GetEnvironmentDamageRate(type);
    if (baseDamageRate <= 0f)
        return;

    // 灾害冲击阶段，建筑区域内扣血减免50%
    if (_currentPhase == GamePhase.DisasterImpact && IsInBuildingZone(charPos))
        baseDamageRate *= 0.5f;

    float dmgMultiplier = _params.DamageMultiplier;
    int hp = isAqua ? _aquaHP : _ignisHP;
    if (hp <= _dyingProtectThreshold)
        dmgMultiplier *= (1f - _dyingProtectReduction);

    float continuousDamage = baseDamageRate * dmgMultiplier * dt;
    int intDmg = Mathf.CeilToInt(continuousDamage);
    if (intDmg > 0)
        DealDamage(type, intDmg);
}
```

**验证**: 地震环境下每2秒扣3HP（而非每帧连续扣血）；陨石靠近落点时概率更高。

### 2.10 TalentSystem 重复叠加Bug修复

**问题**: 每次 `SelectTalent` 调用 `ApplyTalentEffects`，其中 `GetEffectSummary` 返回所有已选天赋的累计值，然后 `ApplyCharacterStats` 用 `+=` 和 `*=` 应用全部累计值，导致 N 次选择后属性被叠加 N 次。

**文件**: `Client/Assets/Scripts/Talent/TalentSystem.cs`

**修复方案**: `ApplyTalentEffects` 改为先重置基础值再应用 summary（全量应用，而非增量叠加）。

```csharp
/// <summary>
/// 应用天赋效果到各系统。
/// 每次调用时先重置基础值，再应用所有天赋的汇总效果。
/// 避免重复叠加问题。
/// </summary>
private void ApplyTalentEffects(CharacterType owner)
{
    TalentEffectSummary summary = GetEffectSummary(owner);

    // 应用角色属性
    ICharacterSystem charSys = ServiceLocator.Get<ICharacterSystem>();
    if (charSys != null)
    {
        ApplyCharacterStats(charSys, owner, summary);
    }

    // 应用庇护参数
    IShelterSystem shelterSys = ServiceLocator.Get<IShelterSystem>();
    if (shelterSys != null)
    {
        ApplyShelterParams(shelterSys, summary);
    }

    // 应用技能系统
    ISkillSystem skillSys = ServiceLocator.Get<ISkillSystem>();
    if (skillSys != null)
    {
        ApplySkillEffects(skillSys, summary);
    }
}

/// <summary>
/// 重置角色属性到基础值后应用天赋汇总。
/// 关键修复：先重置再应用，避免重复叠加。
/// </summary>
private void ApplyCharacterStats(ICharacterSystem charSys, CharacterType type, TalentEffectSummary summary)
{
    CharacterController character = charSys.GetCharacter(type);
    if (character == null || character.Stats == null) return;

    // 重置为基础值（从 CharacterConfig 读取）
    CharacterConfig config = DataManager.Instance.LoadConfig<CharacterConfig>("CharacterConfig");
    if (config != null)
    {
        CharacterStats baseStats = type == CharacterType.Aqua ? config.AquaStats : config.IgnisStats;
        character.Stats.MaxHP = baseStats.MaxHP;
        character.Stats.MoveSpeed = baseStats.MoveSpeed;
        character.Stats.CarryLimit = baseStats.CarryLimit;
    }

    // 应用天赋汇总（全量设置，非叠加）
    character.Stats.MaxHP += summary.HPBonus;
    character.Stats.MoveSpeed *= summary.MoveSpeedMultiplier;
    character.Stats.CarryLimit += summary.CarryLimitBonus;

    // 如果当前HP超过新的MaxHP，截断
    if (character.Stats.CurrentHP > character.Stats.MaxHP)
        character.Stats.CurrentHP = character.Stats.MaxHP;
}

/// <summary>
/// 重置庇护参数到基础值后应用天赋汇总。
/// </summary>
private void ApplyShelterParams(IShelterSystem shelterSys, TalentEffectSummary summary)
{
    // 重置为基础值
    ShelterParams baseParams = new ShelterParams();

    // 应用天赋汇总
    baseParams.MaxEnergy += summary.EnergyMaxBonus;
    baseParams.RecoveryRate *= summary.EnergyRecoveryMultiplier;
    baseParams.ShelterDistance += summary.ShelterDistanceBonus;
    baseParams.DamageMultiplier *= summary.DamageMultiplier;

    // 封顶
    baseParams.RecoveryRate = Mathf.Min(baseParams.RecoveryRate, 20f * 3f); // 基础20×3=60
    baseParams.DamageMultiplier = Mathf.Max(baseParams.DamageMultiplier, 0.1f);

    shelterSys.ModifyParams(baseParams);
}
```

```csharp
private DisasterBase CreateDisaster(DisasterId id, DisasterParams parameters)
{
    // 停止旧灾难
    if (CurrentDisaster != null && CurrentDisaster.IsRunning)

    }

    DisasterBase disaster;
    int category = (int)id / 100;

            break;

            break;
```

### 3.2 SkillSystem 护盾减伤集成到 ShelterSystem

**问题**: `SkillSystem.GetShieldReduction` 方法已实现但 `ShelterSystem.DealDamage` 从未调用它。

**文件**: `Client/Assets/Scripts/Shelter/ShelterSystem.cs`

**修复方案**: 在 `DealDamage` 中查询护盾减伤:

```csharp
public void DealDamage(CharacterType target, int damage)
{
    // 查询护盾减伤
    ISkillSystem skillSys = ServiceLocator.Get<ISkillSystem>();
    if (skillSys != null)
    {
        float shieldReduction = skillSys.GetShieldReduction((byte)target);
        damage = Mathf.RoundToInt(damage * (1f - shieldReduction));
    }

    if (damage <= 0) return;

    if (target == CharacterType.Aqua)
    {
        _aquaHP = Mathf.Max(0, _aquaHP - damage);
        SyncHPToCharacter(target);
        EventBus.Instance.Publish(new PlayerDamagedEvent { playerId = 0, damage = damage });
        if (_aquaHP <= 0)
            EventBus.Instance.Publish(new PlayerDiedEvent { playerId = 0 });
    }
    else
    {
        _ignisHP = Mathf.Max(0, _ignisHP - damage);
        SyncHPToCharacter(target);
        EventBus.Instance.Publish(new PlayerDamagedEvent { playerId = 1, damage = damage });
        if (_ignisHP <= 0)
            EventBus.Instance.Publish(new PlayerDiedEvent { playerId = 1 });
    }
}
```

**验证**: 角色有护盾时受到的伤害减少对应比例。

### 3.3 SkillSystem 被动概率加成集成到 FragmentSystem

**问题**: `SkillSystem.SetPassiveChanceBonus` 存储的值从未被 `FragmentSystem` 读取。

**文件**: `Client/Assets/Scripts/Fragment/FragmentSystem.cs`

**修复方案**: 在 `CheckPassiveSkills` 中查询被动概率加成:

```csharp
private void CheckPassiveSkills(byte playerId, FragmentType collectedType, Vector2 position, bool isJumping, bool isSimultaneous)
{
    ISkillSystem skillSys = ServiceLocator.Get<ISkillSystem>();
    if (skillSys == null)
        return;

    // 基础触发概率
    float triggerChance;
    if (isSimultaneous)
        triggerChance = 1.0f;
    else if (isJumping)
        triggerChance = 0.5f;
    else
        triggerChance = _passiveTriggerChance;

    // 叠加天赋被动概率加成
    float passiveBonus = skillSys.GetPassiveChanceBonus(playerId);
    triggerChance += passiveBonus;
    triggerChance = Mathf.Min01(triggerChance);

    // 寒霜体质
    if (collectedType == FragmentType.IceCrystal
        && skillSys.IsPassiveActive(playerId, PassiveSkillType.FrostAura)
        && Random.Range(0f, 1f) <= triggerChance)
    {
        FreezeNearbyFragments(position, PASSIVE_TRIGGER_RADIUS);
        Debug.Log($"[FragmentSystem] 玩家{playerId} 寒霜体质触发 (概率{triggerChance:F2})");
    }

    // 烈焰体质
    if (collectedType == FragmentType.Lava
        && skillSys.IsPassiveActive(playerId, PassiveSkillType.FlameAura)
        && Random.Range(0f, 1f) <= triggerChance)
    {
        IgniteNearbyFragments(position, PASSIVE_TRIGGER_RADIUS);
        Debug.Log($"[FragmentSystem] 玩家{playerId} 烈焰体质触发 (概率{triggerChance:F2})");
    }
}
```

```csharp

```

在 `SkillSystem` 中实现:
```csharp

{
    return _passiveChanceBonus.TryGetValue(playerId, out float bonus) ? bonus : 0f;
}
```

```csharp
    _timedOutCache.Clear();
    foreach (var kvp in _pendingCollects)

}
```
